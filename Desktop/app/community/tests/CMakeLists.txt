cmake_minimum_required(VERSION 3.25)
project(DeeraftCommunityTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network WebSockets Test)
enable_testing()
add_executable(community-tests CommunityClientTest.cpp ../CommunityClient.cpp ../CommunityClient.h)
target_link_libraries(community-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::WebSockets Qt6::Test)
if(WIN32)
    target_link_libraries(community-tests PRIVATE crypt32)
endif()
add_test(NAME community-tests COMMAND community-tests)
